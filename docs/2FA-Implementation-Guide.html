1. Current Authentication Analysis

1.1 Foundation Already in Place

  • Database schema has twoFactorEnabled and twoFactorSecret fields
  • Dependencies already installed: otplib and qrcode packages
  • Password hashing with bcrypt
  • JWT sessions (currently configured for 5 minutes)

1.2 Security Concerns Identified

  • No 2FA/OTP implementation (database fields exist but unused)
  • Very short session timeout (5 minutes) - too aggressive for production
  • No rate limiting on login attempts
  • No account lockout after failed attempts
  • No email verification system
  • No password reset functionality

2. Implementation Overview

This guide implements Time-Based One-Time Password (TOTP) 2FA using industry-standard authenticator apps (Google Authenticator, Authy, Microsoft Authenticator, etc.)

Note: The database schema is already configured. You only need to implement the application logic and UI components.

Step 2: Create 2FA Setup Page

File: src/app/dashboard/settings/two-factor/page.tsx

'use client'
import { useState, useEffect } from 'react'
import QRCode from 'qrcode'

export default function TwoFactorSetup() {
  const [qrCode, setQrCode] = useState<string>('')
  const [secret, setSecret] = useState<string>('')
  const [verifyCode, setVerifyCode] = useState<string>('')
  const [isEnabled, setIsEnabled] = useState(false)

  useEffect(() => {
    // Generate QR code on mount
    fetch('/api/auth/2fa/setup')
      .then(res => res.json())
      .then(data => {
        setSecret(data.secret)
        QRCode.toDataURL(data.otpauthUrl)
          .then(setQrCode)
      })
  }, [])

  const handleEnable = async () => {
    const response = await fetch('/api/auth/2fa/verify', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ token: verifyCode, secret }),
    })

    if (response.ok) {
      setIsEnabled(true)
      alert('2FA enabled successfully!')
    } else {
      alert('Invalid code. Please try again.')
    }
  }

  return (
    <div className="max-w-md mx-auto p-6">
      <h1 className="text-2xl font-bold mb-6">
        Two-Factor Authentication
      </h1>

      {!isEnabled ? (
        <>
          <p className="mb-4">
            Scan this QR code with your authenticator app
            (Google Authenticator, Authy, etc.)
          </p>

          {qrCode && (
            <img
              src={qrCode}
              alt="2FA QR Code"
              className="mx-auto mb-4"
            />
          )}

          <p className="text-sm text-gray-600 mb-4">
            Manual entry key:
            <code className="bg-gray-100 p-1">{secret}</code>
          </p>

          <div className="mb-4">
            <label className="block mb-2">
              Enter verification code:
            </label>
            <input
              type="text"
              value={verifyCode}
              onChange={(e) => setVerifyCode(e.target.value)}
              className="border p-2 rounded w-full"
              placeholder="000000"
              maxLength={6}
            />
          </div>

          <button
            onClick={handleEnable}
            className="bg-blue-500 text-white px-4 py-2 rounded w-full"
          >
            Enable 2FA
          </button>
        </>
      ) : (
        <div className="text-green-600">
          Two-factor authentication is enabled
        </div>
      )}
    </div>
  )
}

Step 3: Create API Route - Setup Endpoint

File: src/app/api/auth/2fa/setup/route.ts

import { NextResponse } from 'next/server'
import { authenticator } from 'otplib'
import { auth } from '@/auth'
import prisma from '@/app/lib/prisma'

export async function GET() {
  const session = await auth()
  if (!session?.user) {
    return NextResponse.json(
      { error: 'Unauthorized' },
      { status: 401 }
    )
  }

  const secret = authenticator.generateSecret()
  const email = session.user.email!

  const otpauthUrl = authenticator.keyuri(
    email,
    'Stelthion',
    secret
  )

  return NextResponse.json({ secret, otpauthUrl })
}

Step 4: Create API Route - Verify Endpoint

File: src/app/api/auth/2fa/verify/route.ts

import { NextRequest, NextResponse } from 'next/server'
import { authenticator } from 'otplib'
import { auth } from '@/auth'
import prisma from '@/app/lib/prisma'

export async function POST(request: NextRequest) {
  const session = await auth()
  if (!session?.user) {
    return NextResponse.json(
      { error: 'Unauthorized' },
      { status: 401 }
    )
  }

  const { token, secret } = await request.json()

  const isValid = authenticator.verify({ token, secret })

  if (!isValid) {
    return NextResponse.json(
      { error: 'Invalid code' },
      { status: 400 }
    )
  }

  // Save the secret to database
  await prisma.user.update({
    where: { id: session.user.id },
    data: {
      twoFactorEnabled: true,
      twoFactorSecret: secret,
    },
  })

  return NextResponse.json({ success: true })
}

Step 5: Create API Route - Validate Endpoint

File: src/app/api/auth/2fa/validate/route.ts

import { NextRequest, NextResponse } from 'next/server'
import { authenticator } from 'otplib'
import prisma from '@/app/lib/prisma'

export async function POST(request: NextRequest) {
  const { userId, token } = await request.json()

  const user = await prisma.user.findUnique({
    where: { id: userId },
    select: {
      twoFactorSecret: true,
      twoFactorEnabled: true
    },
  })

  if (!user?.twoFactorEnabled || !user.twoFactorSecret) {
    return NextResponse.json(
      { error: 'No 2FA configured' },
      { status: 400 }
    )
  }

  const isValid = authenticator.verify({
    token,
    secret: user.twoFactorSecret,
  })

  return NextResponse.json({ valid: isValid })
}

Step 6: Update Login Action

File: src/app/lib/action.ts

Modify the authenticate function to check for 2FA:

export async function authenticate(
  prevState: string | undefined,
  formData: FormData,
) {
  try {
    const email = formData.get('email') as string
    const password = formData.get('password') as string
    const twoFactorCode = formData.get('twoFactorCode') as string

    // First check credentials
    const user = await prisma.user.findUnique({
      where: { email },
    })

    if (!user) {
      return 'Invalid credentials.'
    }

    const passwordsMatch = await bcrypt.compare(
      password,
      user.password
    )
    if (!passwordsMatch) {
      return 'Invalid credentials.'
    }

    // Check if 2FA is enabled
    if (user.twoFactorEnabled) {
      if (!twoFactorCode) {
        return '2FA_REQUIRED' // Special flag for frontend
      }

      // Verify 2FA code
      const verifyResponse = await fetch(
        '/api/auth/2fa/validate',
        {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            userId: user.id,
            token: twoFactorCode
          }),
        }
      )

      const { valid } = await verifyResponse.json()
      if (!valid) {
        return 'Invalid 2FA code.'
      }
    }

    // Proceed with sign in
    await signIn('credentials', formData)
  } catch (error) {
    if (error instanceof AuthError) {
      switch (error.type) {
        case 'CredentialsSignin':
          return 'Invalid credentials.'
        default:
          return 'Something went wrong.'
      }
    }
    throw error
  }
}

Step 7: Update Login Form

File: src/app/ui/login-form.tsx

Add 2FA code input field:

'use client'
import { Button } from "./button"
import { useActionState, useState } from "react"
import { authenticate } from "../lib/action"
import { useSearchParams } from "next/navigation"

export default function LoginForm() {
  const searchParams = useSearchParams()
  const callbackUrl = searchParams.get('callbackUrl') || '/dashboard'
  const [errorMessage, formAction, isPending] =
    useActionState(authenticate, undefined)
  const [show2FA, setShow2FA] = useState(false)

  // Check if 2FA is required
  if (errorMessage === '2FA_REQUIRED' && !show2FA) {
    setShow2FA(true)
  }

  return (
    <form action={formAction}>
      <h1>Please log in to continue</h1>

      <div>
        <label htmlFor="email">Email</label>
        <input
          type="email"
          name="email"
          id="email"
          required
          className="border p-2 rounded w-full"
          disabled={isPending}
        />
      </div>

      <div>
        <label htmlFor="password">Password</label>
        <input
          type="password"
          name="password"
          id="password"
          required
          className="border p-2 rounded w-full"
          disabled={isPending}
        />
      </div>

      {show2FA && (
        <div>
          <label htmlFor="twoFactorCode">2FA Code</label>
          <input
            type="text"
            name="twoFactorCode"
            id="twoFactorCode"
            required
            className="border p-2 rounded w-full"
            placeholder="000000"
            maxLength={6}
            disabled={isPending}
          />
        </div>
      )}

      {errorMessage && errorMessage !== '2FA_REQUIRED' && (
        <div className="text-red-500 text-sm mt-2">
          {errorMessage}
        </div>
      )}

      <input type="hidden" name="redirectTo" value={callbackUrl}/>
      <Button type="submit" disabled={isPending}>
        {isPending ? 'Logging in...' : 'Log in'}
      </Button>
    </form>
  )
}

4. Additional Security Enhancements

4.1 Rate Limiting

File: src/app/lib/rate-limit.ts

const rateLimitMap = new Map()

export function rateLimit(
  identifier: string,
  limit = 5,
  windowMs = 15 * 60 * 1000
) {
  const now = Date.now()
  const record = rateLimitMap.get(identifier) || {
    count: 0,
    resetTime: now + windowMs
  }

  if (now > record.resetTime) {
    record.count = 0
    record.resetTime = now + windowMs
  }

  record.count++
  rateLimitMap.set(identifier, record)

  return record.count <= limit
}

Add to authenticate function:

import { rateLimit } from './rate-limit'

// At the start of authenticate function:
const allowed = rateLimit(email, 5) // 5 attempts per 15 minutes
if (!allowed) {
  return 'Too many login attempts. Please try again later.'
}
